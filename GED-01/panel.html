Bloquecidio Matematico

Descripción

El juego presenta un fondo estático con estética de plataformas clásicas (cielo, nubes, suelo de ladrillos). En el centro de la pantalla hay un "bloque de interrogación". Al hacer clic en el bloque, se despliega la operación aritmética.

Desarrollo

  • Plataforma: HTML, CSS y JavaScript
  • Lenguaje:JavaScript
  • Gráficos: CSS
  • Objetivo: El Jugador debera responder todas las preguntas matematicas para poder ganar el juego y obtener una puntuacion perfecta

EMPIEZA A JUGAR!!!!!!!

/* =====================================
   ACTUALIZAR JUEGO
===================================== */

function actualizarJuego() {

    if (juegoIniciado == false) {

        return;

    }

    moverJugador();

    aplicarGravedad();

    comprobarColisiones();

    comprobarBloque();

    moverCamara();

    dibujarJugador();

    requestAnimationFrame(
        actualizarJuego
    );

}


/* =====================================
   MOVER JUGADOR
===================================== */

function moverJugador() {

    velocidadJugador = 0;

    if (
        teclas["arrowright"] == true ||
        teclas["d"] == true
    ) {

        velocidadJugador = 5;

    }

    if (
        teclas["arrowleft"] == true ||
        teclas["a"] == true
    ) {

        velocidadJugador = -5;

    }

    posicionJugador =
        posicionJugador + velocidadJugador;

    if (posicionJugador < 20) {

        posicionJugador = 20;

    }

    if (posicionJugador > 1750) {

        posicionJugador = 1750;

    }

    if (
        teclas[" "] == true ||
        teclas["w"] == true ||
        teclas["arrowup"] == true
    ) {

        saltar();

    }

}


/* =====================================
   SALTAR
===================================== */

function saltar() {

    if (puedeSaltar == true) {

        velocidadVertical =
            fuerzaSalto;

        puedeSaltar = false;

    }

}


/* =====================================
   GRAVEDAD
===================================== */

function aplicarGravedad() {

    velocidadVertical =
        velocidadVertical - gravedad;

    posicionVertical =
        posicionVertical + velocidadVertical;

    if (posicionVertical <= 90) {

        posicionVertical = 90;

        velocidadVertical = 0;

        puedeSaltar = true;

    }

}


/* =====================================
   COLISIONES
===================================== */

function comprobarColisiones() {

    var jugadorIzquierda =
        posicionJugador;

    var jugadorDerecha =
        posicionJugador + 45;

    var jugadorAbajo =
        posicionVertical;

    var jugadorArriba =
        posicionVertical + 65;

    var contador = 0;

    while (contador < objetos.length) {

        var objeto =
            objetos[contador];

        var izquierda =
            parseInt(objeto.style.left);

        var abajo =
            parseInt(objeto.style.bottom);

        var ancho =
            objeto.offsetWidth;

        var alto =
            objeto.offsetHeight;

        var derecha =
            izquierda + ancho;

        var arriba =
            abajo + alto;

        var tocaHorizontal =
            jugadorDerecha > izquierda &&
            jugadorIzquierda < derecha;

        var tocaVertical =
            jugadorArriba > abajo &&
            jugadorAbajo < arriba;

        if (
            tocaHorizontal &&
            tocaVertical
        ) {

            if (
                objeto.className.indexOf(
                    "espina"
                ) != -1
            ) {

                reiniciarPosicion();

            } else if (
                objeto.className.indexOf(
                    "cactus"
                ) != -1
            ) {

                reiniciarPosicion();

            } else if (
                objeto.className.indexOf(
                    "roca"
                ) != -1
            ) {

                if (
                    velocidadVertical <= 0 &&
                    jugadorAbajo >= arriba - 25
                ) {

                    posicionVertical =
                        arriba;

                    velocidadVertical = 0;

                    puedeSaltar = true;

                } else {

                    reiniciarPosicion();

                }

            } else if (
                objeto.className.indexOf(
                    "caja"
                ) != -1
            ) {

                if (
                    velocidadVertical <= 0 &&
                    jugadorAbajo >= arriba - 25
                ) {

                    posicionVertical =
                        arriba;

                    velocidadVertical = 0;

                    puedeSaltar = true;

                } else {

                    reiniciarPosicion();

                }

            } else {

                if (
                    velocidadVertical <= 0 &&
                    jugadorAbajo >= arriba - 25
                ) {

                    posicionVertical =
                        arriba;

                    velocidadVertical = 0;

                    puedeSaltar = true;

                }

            }

        }

        contador++;

    }

}

   COMPROBAR BLOQUE

function comprobarBloque() {

    if (preguntaAbierta == true) {

        return;

    }

    var bloque =
        document.getElementById(
            "bloquePregunta"
        );

    var posicionBloque =
        parseInt(bloque.style.left);

            posicionJugador -

}

    if (preguntaAbierta == true) {

/* =====================================
   NUMERO ALEATORIO
===================================== */

function numeroAleatorio(
    minimo,
    maximo
) {

    return Math.floor(
        Math.random() *
        (maximo - minimo + 1)
    ) + minimo;

}


/* =====================================
   GENERAR PREGUNTA
===================================== */

function generarPregunta() {

    var tipoOperacion =
        numeroAleatorio(1, 4);

    var numero1;

    var numero2;

    if (tipoOperacion == 1) {

        numero1 =
            numeroAleatorio(
                10,
                50 + nivelActual * 5
            );

        numero2 =
            numeroAleatorio(
                5,
                40 + nivelActual * 5
            );

        respuestaCorrecta =
            numero1 + numero2;

        document.getElementById(
            "operacion"
        ).innerHTML =
            numero1 +
            " + " +
            numero2 +
            " = ?";

    } else if (tipoOperacion == 2) {

        numero1 =
            numeroAleatorio(
                30,
                100 + nivelActual * 5
            );

        numero2 =
            numeroAleatorio(
                5,
                numero1
            );

        respuestaCorrecta =
            numero1 - numero2;

        document.getElementById(
            "operacion"
        ).innerHTML =
            numero1 +
            " - " +
            numero2 +
            " = ?";

    } else if (tipoOperacion == 3) {

        numero1 =
            numeroAleatorio(
                2,
                8 + Math.floor(
                    nivelActual / 2
                )
            );

        numero2 =
            numeroAleatorio(
                2,
                10
            );

        respuestaCorrecta =
            numero1 * numero2;

        document.getElementById(
            "operacion"
        ).innerHTML =
            numero1 +
            " × " +
            numero2 +
            " = ?";

    } else {

        numero2 =
            numeroAleatorio(
                2,
                10
            );

        var resultadoDivision =
            numeroAleatorio(
                2,
                10 +
                Math.floor(
                    nivelActual / 2
                )
            );

        numero1 =
            numero2 *
            resultadoDivision;

        respuestaCorrecta =
            resultadoDivision;

        document.getElementById(
            "operacion"
        ).innerHTML =
            numero1 +
            " ÷ " +
            numero2 +
            " = ?";

    }

    document.getElementById(
        "numeroPregunta"
    ).innerHTML =
        "Desafío matemático - Nivel " +
        nivelActual;

    generarAlternativas();

}


/* =====================================
   GENERAR ALTERNATIVAS
===================================== */

function generarAlternativas() {

    var alternativas = [];

    alternativas.push(
        respuestaCorrecta
    );

    while (
        alternativas.length < 4
    ) {

        var diferencia =
            numeroAleatorio(
                -10,
                10
            );

        if (diferencia == 0) {

            diferencia = 5;

        }

        var alternativa =
            respuestaCorrecta +
            diferencia;

        if (
            alternativa >= 0 &&
            alternativas.indexOf(
                alternativa
            ) == -1
        ) {

            alternativas.push(
                alternativa
            );

        }

    }

    alternativas.sort(
        function() {

            return Math.random() - 0.5;

        }
    );

    var contenido = "";

    var contador = 0;

    while (contador < 4) {

        contenido =
            contenido +
            '<button class="respuesta" onclick="comprobarRespuesta(' +
            alternativas[contador] +
            ')">' +
            alternativas[contador] +
            '</button>';

        contador++;

    }

    document.getElementById(
        "alternativas"
    ).innerHTML =
        contenido;

    document.getElementById(
        "feedback"
    ).innerHTML = "";

}


/* =====================================
   COMPROBAR RESPUESTA
===================================== */

function comprobarRespuesta(
    respuesta
) {

    var botones =
        document.getElementsByClassName(
            "respuesta"
        );

    var contador = 0;

    while (
        contador < botones.length
    ) {

        botones[contador].disabled =
            true;

        contador++;

    }

    var mensaje;

    if (
        respuesta ==
        respuestaCorrecta
    ) {

        puntaje++;

        mensaje =
            mensajesCorrectos[
                numeroAleatorio(
                    0,
                    mensajesCorrectos.length - 1
                )
            ];

        document.getElementById(
            "feedback"
        ).className =
            "correcto";

        document.getElementById(
            "feedback"
        ).innerHTML =
            "✅ " + mensaje;

    } else {

        mensaje =
            mensajesIncorrectos[
                numeroAleatorio(
                    0,
                    mensajesIncorrectos.length - 1
                )
            ];

        document.getElementById(
            "feedback"
        ).className =
            "incorrecto";

        document.getElementById(
            "feedback"
        ).innerHTML =
            "❌ " + mensaje;

    }

    document.getElementById(
        "puntaje"
    ).innerHTML =
        puntaje;

    setTimeout(
        siguienteNivel,
        1800
    );

}


/* =====================================
   SIGUIENTE NIVEL
===================================== */

function siguienteNivel() {

    document.getElementById(
        "panelPregunta"
    ).style.display =
        "none";

    preguntaAbierta = false;

    if (nivelActual >= 10) {

        terminarJuego();

    } else {

        nivelActual++;

        crearNivel();

    }

}
