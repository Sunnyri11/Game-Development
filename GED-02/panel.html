Reci-Trail

Descripción

El jugador acelera una motocicleta Trail para subir la colina. Al pasar sobre un residuo, se detiene brevemente y debe presionar el botón del color correcto (Amarillo para plástico, Azul para papel, Verde para vidrio) para clasificarlo.

Desarrollo

  • Plataforma: HTML, CSS y JavaScript
  • Lenguaje:JavaScript
  • Gráficos: CSS
  • Objetivo: El Jugador debera recorrer toda la pista con su motocicleta y tendra que limpiar el camino poniendo la basura en sus respectivos contenedores

const terrain = [

    [160, 15],

/* ============================================================
   TERRENO INTERPOLADO
============================================================ */

function terrainHeight(x){

    if(
        x <= terrain[0][0]
    ){
        return terrain[0][1];
    }

    for(
        let i=0;
        i<terrain.length-1;
        i++
    ){

        const a =
            terrain[i];

        const b =
            terrain[i+1];

        if(
            x >= a[0] &&
            x <= b[0]
        ){

            const t =
                (x-a[0]) /
                (b[0]-a[0]);

            const smooth =
                t*t*(3-2*t);

            return (
                a[1] +
                (b[1]-a[1]) *
                smooth
            );
        }
    }

    return terrain[
        terrain.length-1
    ][1];
}


function terrainSlope(x){

    const dx = 2;

    return (
        terrainHeight(x+dx) -
        terrainHeight(x-dx)
    ) /
    (dx*2);
}

function drawRoad(){

    const start =
        Math.max(
            0,
            distance-300
        );

    const end =
        Math.min(
            1900,
            distance+470
        );


    /*
        TERRENO EXTERIOR
    */

    ctx.beginPath();

    for(
        let x=start;
        x<=end;
        x+=4
    ){

        const sx =
            worldToScreenX(x);

        const sy =
            roadY(x);

        if(x===start)
            ctx.moveTo(sx,sy);
        else
            ctx.lineTo(sx,sy);
    }


    for(
        let x=end;
        x>=start;
        x-=4
    ){

        const sx =
            worldToScreenX(x);

            roadY(x);

            ctx.lineTo(sx,sy);

    );

function drawTrash(){

    trash.forEach(
        item => {

            if(item.answered)
                return;


            const diff =
                item.distance -
                distance;


            if(
                diff < -90 ||
                diff > 340
            )
                return;


            const x =
                worldToScreenX(
                    item.distance
                );


            /*
                POSICIÓN REAL DE CARRETERA.
            */

            const y =
                roadCenterY(
                    item.distance
                );


            /*
                Escala de perspectiva.
            */

            const scale =
                Math.max(
                    .7,
                    Math.min(
                        1.15,
                        1 -
                        Math.abs(diff)/850
                    )
                );


            const data =
                trashTypes[
                    item.type
                ];


            ctx.save();


            /*
                sombra
            */

            ctx.fillStyle =
                "rgba(0,0,0,.28)";

            ctx.beginPath();

            ctx.ellipse(
                x,
                y+16*scale,
                22*scale,
                6*scale,
                0,
                0,
                Math.PI*2
            );

            ctx.fill();


            /*
                círculo blanco
            */

            ctx.fillStyle =
                "rgba(255,255,255,.95)";

            ctx.beginPath();

            ctx.arc(
                x,
                y,
                25*scale,
                0,
                Math.PI*2
            );

            ctx.fill();


            ctx.strokeStyle =
                "rgba(20,30,35,.16)";

            ctx.lineWidth=2;

            ctx.stroke();


            /*
                emoji
            */

            ctx.font =
                `${28*scale}px Arial`;

            ctx.textAlign="center";
            ctx.textBaseline="middle";

            ctx.fillText(
                data.emoji,
                x,
                y
            );


            /*
                símbolo
            */

            ctx.font =
                `${11*scale}px Arial`;

            ctx.fillStyle =
                "#28423a";

            ctx.fillText(
                "♻",
                x,
                y-35*scale
            );


            ctx.restore();
        }
    );
}

/*
    La moto ahora está dibujada como
    una composición de muchas piezas.

    - neumáticos
    - llantas
    - suspensión
    - chasis
    - tanque
    - asiento
    - motor
    - escape
    - faro
    - piloto
    - casco
    - manubrio

    Todo se rota junto según la pendiente.
*/

function drawWheel(
    x,
    y,
    r
){

    /*
        Neumático.
    */

    const tire =
        ctx.createRadialGradient(
            x-3,
            y-4,
            2,
            x,
            y,
            r
        );

    tire.addColorStop(
        0,
        "#30383c"
    );

    tire.addColorStop(
        .55,
        "#111518"
    );

    tire.addColorStop(
        1,
        "#050708"
    );

    ctx.fillStyle=tire;

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        r,
        0,
        Math.PI*2
    );

    ctx.fill();


    /*
        Borde.
    */

    ctx.strokeStyle =
        "#0a0d0f";

    ctx.lineWidth=4;

    ctx.stroke();


    /*
        Llanta.
    */

    ctx.fillStyle =
        "#aeb8bb";

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        r*.55,
        0,
        Math.PI*2
    );

    ctx.fill();


    ctx.fillStyle =
        "#485155";

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        r*.37,
        0,
        Math.PI*2
    );

    ctx.fill();


    /*
        Radios.
    */

    ctx.strokeStyle =
        "rgba(230,240,240,.55)";

    ctx.lineWidth=1.3;

    for(
        let i=0;
        i<10;
        i++
    ){

        const a =
            i*Math.PI*2/10;

        ctx.beginPath();

        ctx.moveTo(
            x,
            y
        );

        ctx.lineTo(
            x+
            Math.cos(a)*r*.5,
            y+
            Math.sin(a)*r*.5
        );

        ctx.stroke();
    }


    /*
        Eje.
    */

    ctx.fillStyle =
        "#111719";

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        4,
        0,
        Math.PI*2
    );

    ctx.fill();
}

function openQuestion(item){

    if(
        questionOpen ||
        item.answered
    )
        return;


    currentTrash =
        item;

    questionOpen=true;

    speed=0;


    const data =
        trashTypes[
            item.type
        ];


    document.getElementById(
        "trashEmoji"
    ).textContent =
        data.emoji;

    document.getElementById(
        "trashName"
    ).textContent =
        data.name;


    document.getElementById(
        "questionOverlay"
    ).classList.add(
        "show"
    );
}


/* ============================================================
   RESPONDER
============================================================ */

function answer(type){

    if(
        !questionOpen ||
        !currentTrash
    )
        return;


    /*
        UNA SOLA OPORTUNIDAD.

        Se marca antes de procesar
        para impedir doble respuesta.
    */

    currentTrash.answered=true;


    const correct =
        type ===
        currentTrash.type;


    const data =
        trashTypes[
            currentTrash.type
        ];


    closeQuestion();


    if(correct){

        score += 25;

        /*
            Recuperación.
        */

        energy =
            Math.min(
                100,
                energy + 18
            );


        showMessage(
            "✅",
            "¡Correcto! ♻️",
            data.correct,
            true
        );

    }
    else{

        score =
            Math.max(
                0,
                score - 10
            );


        /*
            Penalización.
        */

        energy =
            Math.max(
                0,
                energy - 22
            );


        showMessage(
            "❌",
            "¡No esta vez!",
            data.wrong +
            " Perdiste 22% de energía.",
            false
        );


        if(
            energy <= 0
        ){

            setTimeout(
                loseGame,
                900
            );
        }
    }


    updateHUD();
}

/* ============================================================
   DETECTAR RESIDUO
============================================================ */

function checkTrash(){

    if(questionOpen)
        return;


    for(
        const item of trash
    ){

        if(item.answered)
            continue;


        const difference =
            item.distance -
            distance;


        /*
            La pregunta aparece cuando
            la moto está prácticamente
            encima del objeto.
        */

        if(
            difference <= 15 &&
            difference >= -12
        ){

            openQuestion(item);

            return;
        }
    }
}


/* ============================================================
   FÍSICA
============================================================ */

function update(dt){

    if(
        !gameRunning ||
        questionOpen
    )
        return;


    const slope =
        terrainSlope(distance);


    /*
        Aceleración.
    */

    if(
        keys.right ||
        keys.left
    ){

        speed +=
            100*dt;

        speed =
            Math.min(
                speed,
                155
            );

    }
    else{

        speed -=
            18*dt;

        speed =
            Math.max(
                speed,
                30
            );
    }


    /*
        SUBIDA.

        Mientras más inclinada,
        más se reduce la velocidad.
    */

    if(
        slope > 0
    ){

        speed -=
            slope *
            14 *
            dt;
    }


    /*
        BAJADA.
    */

    if(
        slope < 0
    ){

        speed +=
            Math.abs(slope) *
            8 *
            dt;
    }


    speed =
        Math.max(
            18,
            Math.min(
                155,
                speed
            )
        );


    /*
        Movimiento.
    */

    distance +=
        speed *
        dt *
        .56;


    /*
        CONSUMO DE ENERGÍA.

        Base:
        conducción constante.

        Extra:
        pendientes.
    */

    let consumption =
        1.45*dt;


    if(
        slope > 0
    ){

        consumption +=
            slope *
            4.2 *
            dt;
    }


    energy -=
        consumption;


    if(
        energy <= 0
    ){

        energy=0;

        updateHUD();

        loseGame();

        return;
    }


    checkTrash();


    /*
        META.
    */

    if(
        distance >= FINISH
    ){

        if(
            score >= 100
        ){

            winGame();

        }
        else{

            loseByPoints();

        }
    }


    updateHUD();
}

function loop(now){

    const dt =
        Math.min(
            (now-lastTime)/1000,
            .033
        );

    lastTime=now;


    update(dt);

    draw();


    requestAnimationFrame(
        loop
    );
}
