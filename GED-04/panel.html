Guardianes del Agua

Descripción

La jugadora recorre diferentes zonas de la ciudad, encuentra problemas relacionados con el agua y debe decidir cómo solucionarlos utilizando recursos limitados. Las decisiones modifican el estado de cada zona.

Desarrollo

  • Plataforma: HTML, CSS y JavaScript
  • Lenguaje:JavaScript
  • Gráficos: CSS
  • Objetivo: Deberas completar todos los minijuegos de la zona para salvarla y asi desbloquear la siguiente. Se necesita de tu ayuda para salvar el mundo
  • Controles:
    Moverse : `A / D` o `← / →`
    Saltar : Espacio
    Seleccionar: E
    Seleccionar opcion/ : Mover las tuberias `Click Izquierdo`

let particles = [];


function createParticles(
    x,
    y,
    color
){

    for(
        let i=0;
        i<16;
        i++
    ){

        particles.push({

            x:x,

            y:y,

            vx:
                (Math.random()-.5)
                *180,

            vy:
                -Math.random()
                *220,

            life:1,

            size:
                3+
                Math.random()*5,

            color:color

        });

    }
}


function updateParticles(dt){

    particles.forEach(
        p => {

            p.x +=
                p.vx*dt;

            p.y +=
                p.vy*dt;

            p.vy +=
                420*dt;

            p.life -=
                dt*1.7;

        }
    );


    particles =
        particles.filter(
            p =>
                p.life > 0
        );
}


function drawParticles(){

    particles.forEach(
        p => {

            ctx.globalAlpha =
                Math.max(
                    0,
                    p.life
                );

            ctx.fillStyle =
                p.color;

            ctx.beginPath();

            ctx.arc(
                p.x-cameraX,
                p.y,
                p.size,
                0,
                Math.PI*2
            );

            ctx.fill();

        }
    );

    ctx.globalAlpha=1;
}

function updateCamera(){

    const target =
        player.x -
        W*.32;


    cameraX +=
        (
            target -

                cameraX,
                worldWidth-W
            )
        );
}


/* =========================================================
   COLISIÓN CON PLATAFORMAS
========================================================= */

function checkPlatformCollision(
    oldY
){

    player.onGround=false;


    for(
        const p of platforms
    ){

        const playerBottom =
            player.y +
            player.height/2;


        const oldBottom =
            oldY +
            player.height/2;


        const horizontal =
            player.x +
            player.width/2 >
            p.x &&

            player.x -
            player.width/2 <
            p.x+p.w;


        const crossing =
            oldBottom <= p.y &&
            playerBottom >= p.y;


        if(
            horizontal &&
            crossing &&
            player.vy >= 0
        ){

            player.y =
                p.y -
                player.height/2;

            player.vy=0;

            player.onGround=true;

            player.coyote=.12;

            return;
        }
    }
}


/* =========================================================
   RECOGER COMIDA
========================================================= */

function checkFood(){

    for(
        const food of foods
    ){

        if(food.collected)
            continue;


        const p =
            platforms[
                food.platform
            ];


        const fx =
            food.x;

        const fy =
            p.y-47;


        const dx =
            player.x -
            fx;

        const dy =
            player.y -
            fy;


        const distance =
            Math.sqrt(
                dx*dx+
                dy*dy
            );


        if(
            distance < 55
        ){

            food.collected=true;


            if(
                food.type ===
                "healthy"
            ){

                /*
                    Comida saludable:
                    aumenta energía.
                */

                energy =
                    Math.min(
                        100,
                        energy+25
                    );


                /*
                    Mejora el salto.
                */

                player.jumpPower =
                    Math.min(
                        900,
                        player.jumpPower+35
                    );


                healthyCount++;


                createParticles(
                    fx,
                    fy,
                    "#65e887"
                );


                showMessage(
                    "🍎",
                    "¡Excelente elección!",
                    food.name +
                    " te da energía y hace tus saltos más fuertes. 💪",
                    true
                );

            }
            else{

                /*
                    Comida chatarra:
                    reduce energía.
                */

                energy =
                    Math.max(
                        5,
                        energy-20
                    );


                /*
                    Reduce salto.
                */

                player.jumpPower =
                    Math.max(
                        500,
                        player.jumpPower-45
                    );


                createParticles(
                    fx,
                    fy,
                    "#ff795e"
                );


                showMessage(
                    "🍔",
                    "¡Cuidado!",
                    food.name +
                    " reduce tu energía y hace tus saltos más pequeños.",
                    false
                );
            }


            updateHUD();
        }
    }
}

window.addEventListener(
    "keydown",
    e => {

        const key =
            e.key.toLowerCase();


        if(
            key === "a" ||
            e.key === "ArrowLeft"
        ){

            keys.left=true;

            e.preventDefault();
        }


        if(
            key === "d" ||
            e.key === "ArrowRight"
        ){

            keys.right=true;

            e.preventDefault();
        }


        if(
            e.code === "Space" ||
            key === "w" ||
            e.key === "ArrowUp"
        ){

            if(
                !keys.jump &&
                gameRunning
            ){

                jump();

            }

            keys.jump=true;

            e.preventDefault();
        }

    }
);


window.addEventListener(

        const key =

        if(
            key === "a" ||
            e.key === "ArrowLeft"
        ){

            keys.left=false;
        }


        if(
            key === "d" ||
            e.key === "ArrowRight"

            keys.right=false;
        }


        if(
            e.code === "Space" ||
            key === "w" ||

            keys.jump=false;
        }

    }
);


/* =========================================================
   SALTO
========================================================= */

function jump(){

    if(
        player.onGround ||
        player.coyote > 0
    ){

        player.vy =
            -player.jumpPower;

        player.onGround=false;

        createParticles(
            player.x,
            player.y+25,
            "#ffffff"
        );
    }
}


/* =========================================================
   ACTUALIZAR JUEGO
========================================================= */

function update(
    dt
){

    if(
        !gameRunning
    )
        return;


    /*
        Temporizador.
    */

    timer -= dt;


    if(
        timer <= 0
    ){

        timer=0;

        winGame();

        return;
    }


    /*
        Movimiento horizontal.
    */

    if(keys.left){

        player.vx =
            -player.speed;

    }
    else if(keys.right){

        player.vx =
            player.speed;

    }
    else{

        player.vx *=
            Math.pow(
                .0005,
                dt
            );
    }


    player.x +=
        player.vx *
        dt;


    /*
        Límites.
    */

    player.x =
        Math.max(
            20,
            Math.min(
                worldWidth-20,
                player.x
            )
        );


    /*
        Gravedad.
    */

    const oldY =
        player.y;


    player.vy +=
        1900 *
        dt;


    player.y +=
        player.vy *
        dt;


    /*
        Coyote time.
    */

    if(
        !player.onGround
    ){

        player.coyote -=
            dt;
    }


    checkPlatformCollision(
        oldY
    );


    checkFood();


    updateCamera();


    updateParticles(
        dt
    );


    /*
        Si cae demasiado,
        pierde.
    */

    if(
        player.y >
        H+180
    ){

        loseGame();

        return;
    }


    updateHUD();
}

    now
){

        Math.min(
            (now-lastTime)/1000,
            .033
        );


    lastTime=now;


    update(dt);

    draw();


    requestAnimationFrame(
        loop
    );
}
