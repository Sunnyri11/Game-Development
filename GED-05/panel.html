Fuera de Pantalla

Descripción

Markus debe realizar un proyecto escolar junto a Sofía, una compañera que comienza a ser víctima de una campaña de ciberbullying. En el grupo de la escuela aparecen memes, rumores y fotografías manipuladas sobre ella. Markus observa cómo sus compañeros comparten el contenido y se ríen, pero al principio decide no intervenir. Con el tiempo, Sofía comienza a aislarse, falta a clases y deja de participar en el proyecto. Markus se da cuenta de que incluso sus propias acciones (como mirar, reaccionar o compartir contenido) pueden contribuir al problema.

/* =========================================================
   TYPEWRITER
========================================================= */

let typingTimer = null;
let isTyping = false;

function typeText(text) {

    if (typingTimer !== null) {
        clearInterval(typingTimer);
    }

    isTyping = true;

    $("#text").text("");

    let index = 0;

    typingTimer = setInterval(
        function() {

            $("#text").text(
                text.substring(
                    0,
                    index
                )
            );

            index++;

            if (index > text.length) {

                clearInterval(
                    typingTimer
                );

                typingTimer = null;

                isTyping = false;
            }

        },
        18
    );

}


/* =========================================================
   SHOW SCENE
========================================================= */

function showScene() {

    if (state.scene >= scenes.length) {

        ending();

        return;
    }

    const scene = scenes[state.scene];

    setBackground(
        scene.bg
    );

    showCharacter(
        scene.char
    );

    if (scene.bg === "night") {

        $("#chapter").text(
            "MARTES · 21:43"
        );

    } else {

        $("#chapter").text(
            "MARTES · 08:17"
        );

    }

    $("#speaker").text(
        scene.speaker
    );

    typeText(
        scene.text
    );

}

        state.scene++;

        showScene();

    }

function selectRoute(route) {

    state.route = route;

    if (route === "ignore") {

        state.ignored = true;
        state.trust -= 15;

    } else if (route === "share") {

        state.shared = true;
        state.trust -= 20;
        state.spreading += 3;

    } else if (route === "defend") {

        state.defended = true;
        state.trust += 15;

    } else if (route === "report") {

        state.reported = true;
        state.trust += 10;

    }

    updateTrust();

    $("#choicePanel").fadeOut(
        300,
        function() {

            state.scene++;

            showScene();

        }
    );

}

        state.trust += trustChange;

        $("#resultTitle").css(
            "color",
            "var(--green)"
        );

    } else {

        state.miniLosses++;

        state.trust -= Math.abs(
            trustChange
        );

        $("#resultTitle").css(
            "color",
            "var(--red)"
        );

    updateTrust();

    $("#resultTitle").text(
        title
    );

    $("#resultDescription").text(
        description
    );

    $("#minigame").hide();

    $("#resultOverlay").css(
        "display",
        "flex"
    );

}

/* =========================================================
   INITIALIZE TETRIS
========================================================= */

function initTetris() {

    tetrisBoard = [];

    tetrisScore = 0;
    tetrisLines = 0;

    tetrisGameOver = false;
    tetrisActive = true;

    for (
        let row = 0;
        row < TETRIS_ROWS;
        row++
    ) {

        tetrisBoard[row] = [];

        for (
            let col = 0;
            col < TETRIS_COLS;
            col++
        ) {

            tetrisBoard[row][col] = null;

        }

    }

    /*
       Añadimos algunos bloques inferiores
       para que el jugador tenga que completar
       líneas realmente.
    */

    for (
        let row = TETRIS_ROWS - 3;
        row < TETRIS_ROWS;
        row++
    ) {

        for (
            let col = 0;
            col < TETRIS_COLS;
            col++
        ) {

            if (Math.random() < 0.52) {

                tetrisBoard[row][col] =
                    "#40517f";

            }

        }

    }

    spawnPiece();

    renderTetris();

    bindTetrisControls();

    tetrisTimer = setInterval(
        function() {

            if (!tetrisActive) {
                return;
            }

            movePieceDown();

        },
        700
    );

}


/* =========================================================
   SPAWN
========================================================= */

function spawnPiece() {

    const randomIndex =
        Math.floor(
            Math.random() *
            TETRIS_PIECES.length
        );

    const template =
        TETRIS_PIECES[randomIndex];

    currentPiece = {
        color: template.color,
        shape: template.shape.map(
            function(row) {
                return row.slice();
            }
        )
    };

    currentX =
        Math.floor(
            (TETRIS_COLS -
            currentPiece.shape[0].length) /
            2
        );

    currentY = 0;

    if (
        collision(
            currentPiece.shape,
            currentX,
            currentY
        )
    ) {

        tetrisGameOver = true;
        tetrisActive = false;

        setTimeout(
            function() {

                miniResult(
                    false,
                    "LA PROPAGACIÓN CONTINÚA",
                    "El tablero se llenó antes de que Markus pudiera frenar la cadena de publicaciones.",
                    10
                );

            },
            300
        );

    }

}


/* =========================================================
   COLLISION
========================================================= */

function collision(
    shape,
    x,
    y
) {

    for (
        let row = 0;
        row < shape.length;
        row++
    ) {

        for (
            let col = 0;
            col < shape[row].length;
            col++
        ) {

            if (!shape[row][col]) {
                continue;
            }

            const boardX = x + col;
            const boardY = y + row;

            if (
                boardX < 0 ||
                boardX >= TETRIS_COLS ||
                boardY >= TETRIS_ROWS
            ) {

                return true;

            }

            if (
                boardY >= 0 &&
                tetrisBoard[boardY][boardX] !== null
            ) {

                return true;

            }

        }

    }

    return false;

}


/* =========================================================
   MOVE LEFT / RIGHT
========================================================= */

function movePieceHorizontal(direction) {

    if (
        !tetrisActive ||
        !currentPiece
    ) {
        return;
    }

    const newX =
        currentX + direction;

    if (
        !collision(
            currentPiece.shape,
            newX,
            currentY
        )
    ) {

        currentX = newX;

        renderTetris();

    }

}


/* =========================================================
   MOVE DOWN
========================================================= */

function movePieceDown() {

    if (
        !tetrisActive ||
        !currentPiece
    ) {
        return;
    }

    const newY =
        currentY + 1;

    if (
        !collision(
            currentPiece.shape,
            currentX,
            newY
        )
    ) {

        currentY = newY;

        renderTetris();

        return;
    }

    lockPiece();

}


/* =========================================================
   HARD DROP
========================================================= */

function hardDrop() {

    if (
        !tetrisActive ||
        !currentPiece
    ) {
        return;
    }

    let safety = 0;

    while (
        !collision(
            currentPiece.shape,
            currentX,
            currentY + 1
        ) &&
        safety < 30
    ) {

        currentY++;

        safety++;

    }

    lockPiece();

}


/* =========================================================
   LOCK PIECE
========================================================= */

function lockPiece() {

    if (!currentPiece) {
        return;
    }

    for (
        let row = 0;
        row < currentPiece.shape.length;
        row++
    ) {

        for (
            let col = 0;
            col < currentPiece.shape[row].length;
            col++
        ) {

            if (!currentPiece.shape[row][col]) {
                continue;
            }

            const boardX =
                currentX + col;

            const boardY =
                currentY + row;

            if (
                boardY >= 0 &&
                boardY < TETRIS_ROWS &&
                boardX >= 0 &&
                boardX < TETRIS_COLS
            ) {

                tetrisBoard[boardY][boardX] =
                    currentPiece.color;

            }

        }

    }

    clearCompletedLines();

    if (tetrisLines >= 3) {

        tetrisActive = false;

        stopTetris();

        miniResult(
            true,
            "PROPAGACIÓN REDUCIDA",
            "Markus consigue frenar parte de la cadena. No puede borrar lo que ya fue visto, pero evita que continúe creciendo.",
            10
        );

        return;
    }

    spawnPiece();

    renderTetris();

}


/* =========================================================
   CLEAR LINES
========================================================= */

function clearCompletedLines() {

    let removed = 0;

    for (
        let row = TETRIS_ROWS - 1;
        row >= 0;
        row--
    ) {

        const complete =
            tetrisBoard[row].every(
                function(cell) {
                    return cell !== null;
                }
            );

        if (complete) {

            tetrisBoard.splice(
                row,
                1
            );

            tetrisBoard.unshift(
                new Array(
                    TETRIS_COLS
                ).fill(null)
            );

            removed++;

            row++;
        }

    }

    if (removed > 0) {

        tetrisLines += removed;

        tetrisScore +=
            removed * 100;

        $("#tetrisScore").text(
            tetrisScore
        );

        $("#tetrisLines").text(
            tetrisLines
        );

    }

}


/* =========================================================
   ROTATION
========================================================= */

function rotateMatrix(matrix) {

    const rows =
        matrix.length;

    const cols =
        matrix[0].length;

    const rotated = [];

    for (
        let col = 0;
        col < cols;
        col++
    ) {

        rotated[col] = [];

        for (
            let row = rows - 1;
            row >= 0;
            row--
        ) {

            rotated[col].push(
                matrix[row][col]
            );

        }

    }

    return rotated;

}


function rotatePiece() {

    if (
        !tetrisActive ||
        !currentPiece
    ) {
        return;
    }

    const rotated =
        rotateMatrix(
            currentPiece.shape
        );

    const tests = [
        0,
        -1,
        1,
        -2,
        2
    ];

    for (
        let i = 0;
        i < tests.length;
        i++
    ) {

        const testX =
            currentX + tests[i];

        if (
            !collision(
                rotated,
                testX,
                currentY
            )
        ) {

            currentPiece.shape =
                rotated;

            currentX =
                testX;

            renderTetris();

            return;

        }

    }

}


/* =========================================================
   RENDER TETRIS
========================================================= */

function renderTetris() {

    const boardElement =
        $("#tetris");

    if (!boardElement.length) {
        return;
    }

    boardElement.empty();

    for (
        let row = 0;
        row < TETRIS_ROWS;
        row++
    ) {

        for (
            let col = 0;
            col < TETRIS_COLS;
            col++
        ) {

            const cell =
                $("<div>")
                    .addClass("tCell");

            const color =
                tetrisBoard[row][col];

            if (color) {

                cell.addClass("block");

                cell.css(
                    "background",
                    color
                );

            }

            /*
               Render de pieza activa.
            */

            if (
                currentPiece &&
                row >= currentY &&
                row <
                    currentY +
                    currentPiece.shape.length &&
                col >= currentX &&
                col <
                    currentX +
                    currentPiece.shape[0].length
            ) {

                const pieceRow =
                    row - currentY;

                const pieceCol =
                    col - currentX;

                if (
                    currentPiece.shape[
                        pieceRow
                    ][
                        pieceCol
                    ]
                ) {

                    cell
                        .removeClass("block")
                        .addClass("active");

                    cell.css(
                        "background",
                        currentPiece.color
                    );

                }

            }

            boardElement.append(
                cell
            );

        }

    }

}

/* =========================================================
   KEYBOARD TETRIS
========================================================= */

function tetrisKeyboard(event) {

    if (
        !$("#minigame").is(":visible") ||
        !tetrisActive
    ) {
        return;
    }

    /*
       Evita que las flechas desplacen
       la página.
    */

    const keys = [
        "ArrowLeft",
        "ArrowRight",
        "ArrowDown",
        "ArrowUp",
        " "
    ];

    if (
        keys.includes(event.key)
    ) {

        event.preventDefault();

    }

    if (event.key === "ArrowLeft") {

        movePieceHorizontal(-1);

    } else if (
        event.key === "ArrowRight"
    ) {

        movePieceHorizontal(1);

    } else if (
        event.key === "ArrowDown"
    ) {

        movePieceDown();

    } else if (
        event.key === "ArrowUp"
    ) {

        rotatePiece();

    } else if (
        event.key === " "
    ) {

        hardDrop();

    }

}

$(document)
    .off("keydown.tetris")
    .on(
        "keydown.tetris",
        tetrisKeyboard
    );

function stopTetris() {

    if (tetrisTimer !== null) {

            tetrisTimer

        tetrisTimer = null;

    }

}
