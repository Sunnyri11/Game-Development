Cookie Dinner

Descripción

El jugador administra un pequeño puesto de comida durante un día. Debe preparar correctamente los pedidos de los clientes, ganar dinero y controlar sus gastos para terminar el día con saldo positivo.

Desarrollo

  • Plataforma: HTML, CSS y JavaScript
  • Lenguaje:JavaScript
  • Gráficos: CSS
  • Objetivo: Recibir un pedido, seleccionar los ingredientes correctos, cocinar el plato y entregarlo a tiempo. Los pedidos correctos generan ganancias; los incorrectos provocan pérdidas.

EMPIEZA A JUGAR!!!!!!!

    const box=

}


/* =====================================================
   MEZCLAR ARRAY
===================================================== */

function shuffle(array){

    for(
        let i=array.length-1;
        i>0;
        i--
    ){

        const j=
            Math.floor(
                Math.random()*(i+1)
            );

        [
            array[i],
            array[j]
        ]=[
            array[j],
            array[i]
        ];
    }

    return array;
}


/* =====================================================
   SELECCIONAR INGREDIENTE
===================================================== */

function toggleIngredient(
    button,
    key
){

    if(selected.includes(key)){

        selected=
            selected.filter(
                x=>x!==key
            );

        button.classList.remove(
            "selected"
        );

    }else{

        selected.push(key);

        button.classList.add(
            "selected"
        );
    }

    document.getElementById("status").textContent=
        `${selected.length} ingrediente(s) seleccionado(s)`;
}


/* =====================================================
   COMPROBAR PEDIDO
===================================================== */

function isCorrectOrder(){

    const recipe=currentCustomer.recipe;

    /* Obligatorios */

    for(const required of recipe.fixed){

        if(!selected.includes(required)){
            return false;
        }
    }

    /* Alternativas */

    for(const group of recipe.alternatives){

        const hasOne=
            group.some(
                ingredient=>
                    selected.includes(ingredient)
            );

        if(!hasOne){
            return false;
        }

        /* No permitir dos del mismo grupo */

        const count=
            group.filter(
                ingredient=>
                    selected.includes(ingredient)
            ).length;

        if(count!==1){
            return false;
        }
    }

    /*
       No puede haber ingredientes
       que no formen parte del pedido.
    */

    const allowed=new Set(
        recipe.fixed.concat(
            recipe.alternatives.flat()
        )
    );

    for(const ingredient of selected){

        if(!allowed.has(ingredient)){
            return false;
        }
    }

    return true;
}


/* =====================================================
   COCINAR

function cookOrder(){

    if(
        !gameRunning ||
        !currentCustomer
    ){
        return;
    }

    const correct=isCorrectOrder();

    if(!correct){

        wrongRecipe();

        return;
    }

    completeOrder();
}


/* =====================================================

function wrongRecipe(){

    continueAfterOrder();
}


/* =====================================================
   PEDIDO CORRECTO
===================================================== */

function completeOrder(){

    const recipe=
        currentCustomer.recipe;

    let cost=0;

    selected.forEach(
        key=>{
            cost+=INGREDIENTS[key].cost;
        }
    );

    let tip=0;

    if(currentCustomer.time>=40){

        tip=3;

    }else if(currentCustomer.time>=20){

        tip=1;
    }

    const earned=
        recipe.price+tip;

    const profit=
        earned-cost;

    money+=profit;

    income+=earned;

    expenses+=cost;

    completed++;

    combo++;

    document.getElementById("lastCost").textContent=
        "$"+cost;

    /*
       Mensaje especial si tomó una
       decisión económica.
    */

    const minimumPossible=
        calculateMinimumCost(recipe);

    if(cost===minimumPossible){

        showMessage(

            "💚 ¡Excelente decisión!",

            `Vendiste por $${earned}, gastaste solo $${cost} `
            +`y ganaste $${profit}. Elegiste una opción económica.`,

            "good"

        );

    }else if(cost>minimumPossible+3){

        showMessage(

            "💸 Podías gastar menos",

            `Vendiste por $${earned}, pero gastaste $${cost}. `
            +`Tu ganancia fue $${profit}. `
            +`Había una alternativa más económica.`,

            "warning"

        );

    }else{

        showMessage(

            "🍽️ ¡Pedido entregado!",

            `Ingresaste $${earned}, gastaste $${cost} `
            +`y tu ganancia fue $${profit}.`,

            "good"

        );
    }

    if(tip>0){

        setTimeout(()=>{

            showMessage(

                "⭐ ¡Propina!",

                `El cliente quedó satisfecho con la rapidez. `
                +`Recibiste $${tip} extra.`,

                "good"

            );

        },700);
    }

    if(combo===3){

        money+=3;

        showMessage(

            "🔥 ¡COMBO DE 3!",

            "Tres pedidos correctos seguidos. "
            +"Recibiste un bono de $3.",

            "good"

        );
    }

    currentCustomer=null;

    selected=[];

    updateUI();

    renderCustomers();

    continueAfterOrder();
}


/* =====================================================
   COSTO MÍNIMO POSIBLE
===================================================== */

function calculateMinimumCost(recipe){

    let total=0;

    recipe.fixed.forEach(
        key=>{
            total+=INGREDIENTS[key].cost;
        }
    );

    recipe.alternatives.forEach(
        group=>{

            let cheapest=Infinity;

            group.forEach(
                key=>{

                    cheapest=
                        Math.min(
                            cheapest,
                            INGREDIENTS[key].cost
                        );

                }
            );

            total+=cheapest;
        }
    );

    return total;
}


/* =====================================================
   CONTINUAR DESPUÉS DEL PEDIDO
===================================================== */

function continueAfterOrder(){

    if(completed>=totalOrders){

        setTimeout(
            endDay,
            800
        );

        return;
    }

    /*
       El siguiente cliente pasa inmediatamente.
    */

    setTimeout(
        selectNextCustomer,
        600
    );

    /*
       Después de cada pedido
       intentamos traer otro cliente.
    */

    setTimeout(
        tryAddWaitingCustomer,
        2200
    );
}


/* =====================================================
   NUEVO CLIENTE
===================================================== */

function tryAddWaitingCustomer(){

    if(!gameRunning){
        return;
    }

    if(
        customers.length<maxWaiting()
    ){

        addCustomer();

        showMessage(

            "👋 ¡Nuevo cliente!",

            "Se puso en la fila. "
            +"Su paciencia también está disminuyendo.",

            "warning"

        );
    }

    renderCustomers();
}


/* =====================================================
   TEMPORIZADOR
===================================================== */

function gameTick(){

    if(!gameRunning){
        return;
    }

    /*
       CLIENTE PRINCIPAL
    */

    if(currentCustomer){

        currentCustomer.time--;
    }


    /*
       CLIENTES ESPERANDO
    */

    customers.forEach(
        customer=>{
            customer.time--;
        }
    );


    /*
       PRINCIPAL
    */

    if(
        currentCustomer &&
        currentCustomer.time<=0
    ){

        const leaving=
            currentCustomer;

        currentCustomer=null;

        combo=0;

        customerLeft(
            leaving,
            true
        );

        completed++;

        selected=[];

        setTimeout(
            selectNextCustomer,
            500
        );
    }


    /*
       ESPERANDO
    */

    for(
        let i=customers.length-1;
        i>=0;
        i--
    ){

        const customer=
            customers[i];

        if(customer.time<=0){

            customerLeft(
                customer,
                false
            );

            customers.splice(
                i,
                1
            );

        }
    }

    renderCustomers();

    updateUI();
}


/* =====================================================
   CLIENTE SE ENFADA
===================================================== */

function customerLeft(
    customer,
    wasMain
){

    /*
       La pérdida ocurre aunque
       el dinero pueda quedar negativo.
    */

    const penalty=5;

    money-=penalty;

    expenses+=penalty;

    combo=0;

    showMessage(

        "😡 ¡Cliente enfadado!",

        `${customer.person[1]} esperó demasiado y se fue. `
        +`Perdiste $${penalty}.`,

        "bad"

    );

    if(wasMain){

        document.getElementById("status").textContent=
            "😡 El cliente se fue";
    }
}

/* =====================================================
   SIGUIENTE DÍA
===================================================== */

function nextDay(){

    document.getElementById(
        "dayModal"
    ).classList.remove("show");

    day++;

    if(day>3){

        document.getElementById(
            "winModal"
        ).classList.add("show");

        return;
    }


    /*
       10 → 15 → 20
    */

    if(day===1){

        totalOrders=10;

    }else if(day===2){

        totalOrders=15;

    }else{

        totalOrders=20;
    }


    completed=0;

    income=0;

    expenses=0;

    combo=0;

    customers=[];

    currentCustomer=null;

    selected=[];

    gameRunning=true;


    clearArrivalTimers();


    /*
       Un cliente inicial.
    */

    addCustomer();

    selectNextCustomer();


    clearInterval(timer);

    timer=
        setInterval(
            gameTick,
            1000
        );


    scheduleArrivals();

    updateUI();


    showMessage(

        "🌞 ¡Día "+day+"!",

        day===2

        ?

        "Ahora tendrás 15 pedidos y aparecerán más clientes en espera."

        :

        "¡Día final! Tendrás 20 pedidos y pueden aparecer hasta 3 clientes esperando.",

        "warning"

    );
}


/* =====================================================
   APARICIÓN PROGRESIVA
===================================================== */

function scheduleArrivals(){

    clearArrivalTimers();

    const arrivals=
        maxWaiting();

    const interval=
        arrivalSpeed();


    for(
        let i=0;
        i<arrivals;
        i++
    ){

        const timeout=
            setTimeout(
                ()=>{

                    if(
                        gameRunning &&
                        customers.length<maxWaiting() &&
                        completed<totalOrders
                    ){

                        addCustomer();

                        showMessage(

                            "👋 ¡Nuevo cliente!",

                            "Recuerda: aunque esté esperando, "
                            +"su tiempo también está corriendo.",

                            "warning"

                        );
                    }

                },
                interval*(i+1)
            );

        arrivalTimers.push(timeout);
    }
}


/* =====================================================
   LIMPIAR LLEGADAS
===================================================== */

function clearArrivalTimers(){

    arrivalTimers.forEach(
        timeout=>{
            clearTimeout(timeout);
        }
    );

    arrivalTimers=[];
}


/* =====================================================
   DERROTA
===================================================== */

function loseGame(){

    /*
       Ya no se usa para detener el juego
       simplemente por estar negativo.

       La derrota se comprueba al FINAL
       del día.

       Esta función queda como protección
       para eventos externos.
    */

    gameRunning=false;

    clearInterval(timer);

    clearArrivalTimers();

    document.getElementById(
        "loseModal"
    ).classList.add("show");
}
